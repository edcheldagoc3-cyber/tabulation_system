namespace App\Controllers;

use App\Core\Controller;
use App\Core\Auth;
use App\Models\ScoreModel;
use App\Models\CategoryModel;
use App\Core\TabulationEngine;

class ScoreController extends Controller {

    /**
     * GET /score/form?contestant_id=X&category_id=Y
     * Displays the scoring form for a judge (used by the view).
     */
    public function form() {
        $user = Auth::requireRole('judge');
        $contestantId = $_GET['contestant_id'] ?? null;
        $categoryId = $_GET['category_id'] ?? null;

        if (!$contestantId || !$categoryId) {
            die("Missing contestant or category ID.");
        }

        // FR7 & FR10: Judge must be assigned to this category
        if (!Auth::judgeOwnsCategoryAssignment($categoryId)) {
            die("You are not assigned to this category.");
        }

        $category = ScoreModel::getCategoryDetails($categoryId);
        $criteria = ScoreModel::getCriteriaByCategory($categoryId);
        $contestants = ScoreModel::getContestantsByCategory($categoryId);
        $assignment = ScoreModel::getJudgeAssignment($user->id, $categoryId);

        if (empty($criteria)) {
            $this->json(['error' => 'No criteria have been configured for this category yet.'], 422);
        }

        // Return as JSON (your frontend JS can consume this)
        $this->json([
            'message' => 'Scoring form loaded',
            'category' => $category,
            'criteria' => $criteria,
            'contestants' => $contestants,
            'assignment_id' => $assignment->id ?? null
        ]);
    }

    /**
     * POST /score/submit
     * Handles score submission, validation, locking, and auto-tabulation.
     */
    public function submit() {
        // 1. Authenticate – only judges can submit
        $user = Auth::requireRole('judge');

        if (!function_exists('verify_csrf_token') || !verify_csrf_token()) {
            $this->json(['error' => 'Invalid security token. Please refresh the page and try again.'], 419);
        }

        // 2. Get JSON payload
        $data = json_decode(file_get_contents('php://input'), true);
        if (!is_array($data)) {
            $this->json(['error' => 'Invalid JSON payload'], 400);
        }
        $contestantId = $data['contestant_id'] ?? null;
        $categoryId = $data['category_id'] ?? null;
        $scores = $data['scores'] ?? []; // [criteria_id => score_value]
        $isDraft = (bool)($data['is_draft'] ?? false);

        // 3. Validate required fields
        if (!$contestantId || !$categoryId || empty($scores)) {
            $this->json(['error' => 'Missing required fields (contestant_id, category_id, scores)'], 400);
        }

        // 4. Authorization – judge must be assigned to this category (FR7, FR10)
        if (!Auth::judgeOwnsCategoryAssignment($categoryId)) {
            $this->json(['error' => 'Unauthorized – you are not assigned to this category'], 403);
        }

        // 5. Get judge assignment record
        $assignment = ScoreModel::getJudgeAssignment($user->id, $categoryId);
        if (!$assignment) {
            $this->json(['error' => 'Assignment not found for this judge and category'], 404);
        }

        // 6. Validate scores against criteria min/max (FR8 validation)
        $criteria = ScoreModel::getCriteriaByCategory($categoryId);
        if (empty($criteria)) {
            $this->json(['error' => 'No criteria have been configured for this category yet.'], 422);
        }
        foreach ($criteria as $c) {
            if (!isset($scores[$c->id])) {
                $this->json(['error' => "Missing score for criterion: {$c->name}"], 400);
            }
            if ($scores[$c->id] < $c->min_score || $scores[$c->id] > $c->max_score) {
                $this->json([
                    'error' => "Score for '{$c->name}' must be between {$c->min_score} and {$c->max_score}"
                ], 400);
            }
        }

        // 7. Save scores and lock them (FR9 – scores locked after submission if not draft)
        try {
            foreach ($scores as $critId => $score) {
                ScoreModel::saveScore($assignment->id, $contestantId, $critId, $score);
            }

            if (!$isDraft) {
                // Lock all scores for this judge+contestant combination
                ScoreModel::lockScoresForContestant($assignment->id, $contestantId);

                // ========== STEP 7: AUTO-TABULATION ==========
                // After scores are saved, recompute the entire category rankings.
                $this->recomputeCategory($categoryId);

                $this->json(['message' => 'Scores submitted, locked, and tabulation updated successfully']);
            } else {
                $this->json(['message' => 'Draft scores saved successfully']);
            }

        } catch (\Exception $e) {
            $this->json(['error' => $e->getMessage()], 500);
        }
    }

    /**
     * Recomputes rankings for a category using the TabulationEngine.
     * This is called automatically after every score submission (Step 7).
     */
    private function recomputeCategory($categoryId) {
        // 1. Fetch category details
        $category = ScoreModel::getCategoryDetails($categoryId);
        if (!$category) {
            throw new \Exception("Category not found.");
        }

        // 2. Fetch all locked scores for this category
        $scoresRaw = ScoreModel::getScoresForCategory($categoryId);
        if (empty($scoresRaw)) {
            // No scores yet – nothing to compute
            return;
        }

        // 3. Fetch criteria and their weights
        $criteria = ScoreModel::getCriteriaByCategory($categoryId);
        $criteriaWeights = [];
        foreach ($criteria as $c) {
            $criteriaWeights[$c->id] = ['weight' => $c->weight_percent / 100];
        }

        // 4. Format scores for TabulationEngine:
        //    [judge_id][contestant_id][criteria_id] = score_value
        $scoresFormatted = [];
        foreach ($scoresRaw as $s) {
            $scoresFormatted[$s->judge_id][$s->contestant_id][$s->criteria_id] = $s->score_value;
        }

        // 5. Run the engine
        $engineResult = TabulationEngine::compute(
            $scoresFormatted,
            $criteriaWeights,
            $category->computation_type
        );

        // 6. Save results (insert or update)
        foreach ($engineResult as $contestantId => $result) {
            ScoreModel::saveResult(
                $categoryId,
                $contestantId,
                $result['score'],
                $result['rank']
            );
        }
    }

namespace App\Controllers;

use App\Core\Auth;

use App\Models\CategoryModel;
use App\Models\JudgeAssignmentModel;
use App\Models\ResultModel;
use App\Models\ScoreModel;

class TabulatorController extends Controller {

    public function index() {
        Auth::requireRole('tabulator');

        $categories = CategoryModel::getAllWithProgress();
        $summary = [
            'categories_total' => count($categories),
            'complete' => count(array_filter($categories, static function ($category) {
                return (int)($category->progress_percent ?? 0) >= 100;
            })),
            'released' => count(array_filter($categories, static function ($category) {
                return !empty($category->is_released);
            })),
        ];

        $this->view('tabulator/dashboard', [
            'categories' => $categories,
            'summary' => $summary,
        ]);
    }

    public function release($categoryId) {
        Auth::requireRole('tabulator');
        $this->assertCsrf();

        $category = CategoryModel::getById($categoryId);
        if (!$category) {
            http_response_code(404);
            die('Category not found.');
        }

        $criteria = ScoreModel::getCriteriaByCategory($categoryId);
        $scoresRaw = ScoreModel::getScoresForCategory($categoryId);

        $scoresFormatted = [];
        foreach ($scoresRaw as $scoreRow) {
            $scoresFormatted[$scoreRow->judge_id][$scoreRow->contestant_id][$scoreRow->criteria_id] = $scoreRow->score_value;
        }

        if (!empty($scoresFormatted)) {
            $criteriaWeights = [];
            foreach ($criteria as $criterion) {
                $criteriaWeights[$criterion->id] = ['weight' => ((float)$criterion->weight_percent) / 100];
            }

            $computed = TabulationEngine::compute($scoresFormatted, $criteriaWeights, $category->computation_type);
            foreach ($computed as $contestantId => $result) {
                ResultModel::saveResult($categoryId, $contestantId, $result['score'], $result['rank']);
            }
        }

        ResultModel::releaseCategory($categoryId, Auth::user()->id);
        $this->redirect('/tabulator');
    }

    private function assertCsrf() {
        if (!function_exists('verify_csrf_token') || !verify_csrf_token()) {
            http_response_code(419);
            die('Invalid security token. Please refresh and try again.');
        }
    }

use App\Core\Controller;

class ViewerController extends Controller {

    public function index() {
        $events = EventModel::getAll();

        $eventId = isset($_GET['event_id']) ? (int)$_GET['event_id'] : 0;
        $categoryId = isset($_GET['category_id']) ? (int)$_GET['category_id'] : 0;

        if ($eventId <= 0) {
            foreach ($events as $event) {
                $releasedCategories = ResultModel::getReleasedCategoriesByEvent($event->id);
                if (!empty($releasedCategories)) {
                    $eventId = (int)$event->id;
                    break;
                }
            }
        }

        if ($eventId <= 0 && !empty($events)) {
            $eventId = (int)$events[0]->id;
        }

        $categories = $eventId > 0 ? CategoryModel::getByEvent($eventId) : [];
        $releasedCategories = [];
        foreach ($categories as $category) {
            $categoryResults = ResultModel::getReleasedByCategory($category->id);
            if (!empty($categoryResults)) {
                $releasedCategories[] = $category;
            }
        }

        if ($categoryId <= 0) {
            $categoryId = !empty($releasedCategories) ? (int)$releasedCategories[0]->id : (!empty($categories) ? (int)$categories[0]->id : 0);
        }

        $results = $categoryId > 0 ? ResultModel::getReleasedByCategory($categoryId) : [];

<?php
namespace App\Core;

class TabulationEngine {

    public static function compute($scores, $criteria, $type) {
        $judgeAverages = [];

        foreach ($scores as $judgeId => $contestants) {
            foreach ($contestants as $contestantId => $criteriaScores) {
                if (!isset($judgeAverages[$contestantId])) {
                    $judgeAverages[$contestantId] = [];
                }

                if ($type === 'rank_based') {
                    $judgeTotals = array_sum($criteriaScores);
                    $judgeAverages[$contestantId][$judgeId] = $judgeTotals;
                    continue;
                }

                $total = 0.0;
                foreach ($criteriaScores as $critId => $score) {
                    $weight = $criteria[$critId]['weight'] ?? 1;
                    if ($type === 'weighted_criteria') {
                        $total += ((float)$score) * ((float)$weight);
                    } else {
                        $total += (float)$score;
                    }
                }
                $judgeAverages[$contestantId][$judgeId] = $total;
            }
        }

        $finalScores = [];
        foreach ($judgeAverages as $contestantId => $judgeTotals) {
            $count = count($judgeTotals);
            if ($count === 0) {
                continue;
            }
            $finalScores[$contestantId] = array_sum($judgeTotals) / $count;
        }

        $ranked = [];
        $sorted = $finalScores;
        if ($type === 'rank_based') {
            asort($sorted); // lower average rank is better
        } else {
            arsort($sorted); // higher score is better
        }

        $rank = 1;
        $position = 0;
        $previousScore = null;
        foreach ($sorted as $contestantId => $score) {
            $position++;
            if ($previousScore !== null && (float)$score === (float)$previousScore) {
                // Keep the same rank for ties.
            } else {
                $rank = $position;
            }

            $ranked[$contestantId] = [
                'score' => $score,
                'rank' => $rank,
            ];
            $previousScore = $score;
        }

        return $ranked;
    }
}

     * Update a user's details.
     */

        $db = self::getDB();
        $fields = [];
        $params = [];
        if (isset($data['full_name'])) {
            $fields[] = "full_name = ?";
            $params[] = $data['full_name'];
        }
        if (isset($data['email'])) {
            $fields[] = "email = ?";
            $params[] = $data['email'];
        }
        if (isset($data['role'])) {

function showToast(message) {
    const toast = document.getElementById('toast');
    toast.textContent = message;
    toast.classList.add('show');
    setTimeout(() => toast.classList.remove('show'), 2600);

function openScoring(contestantId) {
    currentContestantId = contestantId;
    const contestant = contestants.find(c => Number(c.id) === Number(contestantId));
    document.getElementById('modalTitle').textContent = contestant ? contestant.name : 'Contestant';
    document.getElementById('modalHint').textContent = contestant ? `Scoring ${contestant.name}` : 'Set the scores for this contestant.';
    const saved = existingScores[contestantId] || {};
    const container = document.getElementById('criteriaContainer');
    container.innerHTML = '';

    criteria.forEach(criterion => {
        const min = Number(criterion.min_score ?? 0);
        const max = Number(criterion.max_score ?? 100);
        const savedValue = saved[criterion.id] !== undefined ? saved[criterion.id] : ((min + max) / 2);
        const valueId = `score_${criterion.id}`;
        const block = document.createElement('div');
        block.className = 'criterion';
        block.innerHTML = `
            <header>
                <div>
                    <label>${escapeHtml(criterion.name)}</label><br>
                    <small>Weight: ${escapeHtml(String(criterion.weight_percent ?? '0'))}%</small>
                </div>
                <div class="score-value" id="${valueId}_display">${formatScore(savedValue)}</div>
            </header>
            <input type="range"
                min="${min}"
                max="${max}"
                step="0.01"
                value="${savedValue}"
                data-criterion-id="${criterion.id}"
                oninput="updateCriterionValue(this)">
            <div class="slider-meta">
                <span>${min}</span>
                <span>${max}</span>
            </div>
        `;
        container.appendChild(block);
    });

    document.getElementById('scoreModal').classList.add('open');
    document.getElementById('scoreModal').setAttribute('aria-hidden', 'false');
}

function closeModal() {
    document.getElementById('scoreModal').classList.remove('open');
    document.getElementById('scoreModal').setAttribute('aria-hidden', 'true');
}

function updateCriterionValue(input) {
    const criterionId = input.getAttribute('data-criterion-id');
    const display = document.getElementById(`score_${criterionId}_display`);
    if (display) display.textContent = formatScore(input.value);
}

function gatherScores() {
    const scores = {};
    document.querySelectorAll('#criteriaContainer input[type="range"]').forEach(input => {
        scores[input.getAttribute('data-criterion-id')] = Number(parseFloat(input.value).toFixed(2));
    });
    return scores;
}

async function submitScores(isDraft) {
    if (!currentContestantId) return;
    const response = await fetch('/score/submit', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'X-CSRF-Token': csrfToken
        },
        body: JSON.stringify({
            contestant_id: currentContestantId,
            category_id: categoryId,
            scores: gatherScores(),
            is_draft: isDraft
        })
    });

    const payload = await response.json();
    if (!response.ok || payload.error) {
        throw new Error(payload.error || 'Unable to save scores.');
    }
    return payload;
}

async function saveDraft() {
    try {
        await submitScores(true);
        showToast('Draft saved successfully.');
        closeModal();
        setTimeout(() => window.location.reload(), 600);
    } catch (error) {
        alert(error.message);
    }
}

async function submitFinal() {
    if (!confirm('Submit final scores for this contestant? This will lock the scores.')) {
        return;
    }
    try {
        await submitScores(false);
        showToast('Final scores submitted.');
        closeModal();
        setTimeout(() => window.location.reload(), 600);
    } catch (error) {
        alert(error.message);
    }
}

document.getElementById('scoreModal').addEventListener('click', function(event) {
    if (event.target === this) closeModal();
});

function formatScore(value) {
    const num = Number(value);
    return Number.isInteger(num) ? String(num) : num.toFixed(2);
}

function escapeHtml(value) {
    return String(value)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#039;');
}

namespace Config;

use PDO;
use PDOException;

class Database {
    private static $instance = null;
    private $conn;

    private function __construct() {
        $host = $_ENV['DB_HOST'] ?? 'localhost';
        $port = $_ENV['DB_PORT'] ?? '3306';
        $dbname = $_ENV['DB_DATABASE'] ?? 'ustp_tabulation';
        $username = $_ENV['DB_USERNAME'] ?? 'root';
        $password = $_ENV['DB_PASSWORD'] ?? '';

        try {
            $dsn = "mysql:host=$host;port=$port;dbname=$dbname;charset=utf8mb4";
            $this->conn = new PDO($dsn, $username, $password);
            $this->conn->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
            $this->conn->setAttribute(PDO::ATTR_DEFAULT_FETCH_MODE, PDO::FETCH_OBJ);
        } catch (PDOException $e) {
            die("Database Connection Failed: " . $e->getMessage());
        }
    }

    public static function getConnection(): PDO {
        if (self::$instance === null) {
            self::$instance = new self();
        }
        return self::$instance->conn;
    }
